\section{Introduction} 
\par
This document presents a detailed overview of the Elaume formalism.
The first section presents the derivations and assumptions used to obtain the kickmap potential,
then a discussion about the symplecticity of the kickmaps is presented and an application of the formalism for
a sinusoidal field and for an adjustable-phase undulator is shown, presenting also the effects predicted on the beam dynamics.

\section{Ellaume formalism}
\subsection{Equations of motion}

The derivations presented in this section are based on the work of Ellaume \cite{elleaume1992approach} and are 
detailed on the following tecnical report from CLS \cite{wurtz2014kickmaps}.

Using SIRIUS coordinate system, the equations of motion for a particle in a magnetic field can be expressed as:
\begin{equation}
    \frac{d\vec{p}}{dt} = q\vec{v} \times \vec{B}
    \label{eq:lorentz_force}
\end{equation}
$\vec{p}$ is the momentum of the electron, $q= -e$ is the electron's charge, $\vec{v}$ is its velocity, and $\vec{B}$ is
the magnetic field along the electron's path.
The velocity can be written as:
\begin{equation}
    \vec{v} = \frac{dx}{dt}\hat{x} + \frac{dy}{dt}\hat{y} + \frac{ds}{dt}\hat{s} = 
    \left(\frac{ds}{dt}\right)\left(\frac{dx}{ds}\hat{x} + \frac{dy}{ds}\hat{y} + \hat{s}\right)
    \label{eq:velocity}
\end{equation}
defining
\begin{equation}
    x^{\prime} \coloneqq \frac{dx}{ds}, \quad y^{\prime} \coloneqq \frac{dy}{ds}
    \label{eq:prime}
\end{equation}
and assuming the electorn is ultra-relativitic, we have:
\begin{equation}
    c^2 = \vec{v}^2 = \left(\frac{ds}{dt}\right)^2(x^{\prime 2} + y^{\prime 2} + 1)
\end{equation}
then, it is found the following expression for $ds/dt$:
\begin{equation}
    \frac{ds}{dt} = \frac{c}{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}}
\end{equation}
which gives the expression for the velocity:
\begin{equation}
    \vec{v} = \frac{c}{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}}\left(x^{\prime}\hat{x} + y^{\prime}\hat{y} + \hat{s}\right)
\end{equation}
Substituting this expression into the Lorentz force equation, we have:
\begin{equation}
    \frac{d\vec{p}}{dt} = \frac{-ec}{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}}\left[(y^{\prime} B_s-B_y)\hat{x} -(x^{\prime} B_s-B_x)\hat{y} + (x^{\prime} B_y-y^{\prime} B_x)\hat{s}\right]
    \label{eq:lorentz_force_expanded}
\end{equation}
Keeping the result of equation \eqref{eq:lorentz_force_expanded}, now let's express the momentum of a relativistic electron:
\begin{equation}
    \vec{p} = \gamma m_e \vec{v} = \frac{\gamma m_e c}{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}}\left(x^{\prime}\hat{x} + y^{\prime}\hat{y} + \hat{s}\right)
\end{equation}
and it's time derivative:
\begin{equation}
    \frac{d\vec{p}}{dt} = \frac{ds}{dt}\frac{d\vec{p}}{ds} = \frac{\gamma m_e c^2}{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}}
    \left[-(x^{\prime 2}+y^{\prime 2}+1)^{-3/2}(x^{\prime} x^{\prime\prime} + y^{\prime} y^{\prime\prime})(x^{\prime}\hat{x} + y^{\prime}\hat{y} + \hat{s})
    + \frac{x^{\prime\prime}\hat{x} + y^{\prime\prime}\hat{y}}{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}}\right]
    \label{eq:momentum_derivative}
\end{equation}
Then, using the magnetic rigidity $R = \gamma m_e c / e$ and equating the $\hat{s}$ component of equations \eqref{eq:lorentz_force_expanded}
and \eqref{eq:momentum_derivative}, we have:
\begin{equation}
    \frac{x{^\prime}x^{\prime\prime} + y^{\prime}y^{\prime\prime}}{\left(x^{\prime 2} + y^{\prime 2 } + 1\right)^{3/2}} = \frac{1}{R}\left(x^{\prime}B_y - y^{\prime}B_x\right)
\end{equation}
substituting equation above into equation \eqref{eq:momentum_derivative}, the remaining two vector components are related by
\begin{equation}
    \frac{-1}{R}\left(x^{\prime} B_y - y^{\prime} B_x \right)\left(x^{\prime}\hat{x} + y^{\prime}\hat{y}\right)
     + \frac{x^{\prime \prime}\hat{x} + y^{\prime\prime}\hat{y}}{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}} = \frac{-1}{R}
     \left[\left(y^{\prime}B_s - B_y\right)\hat{x} + \left(x^{\prime}B_s - B_x\right)\right]
\end{equation}
Solving this equation for $x^{\prime\prime}$ and $y^{\prime\prime}$, we have the following equations of motion:
\begin{equation}
    \begin{aligned}
        x^{\prime\prime} = -\frac{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}}{R}\left[x^{\prime}y^{\prime}B_x -\left(x^{\prime 2} + 1\right)B_y +y^{\prime}Bs\right]\\
        y^{\prime\prime} = -\frac{\sqrt{x^{\prime 2} + y^{\prime 2} + 1}}{R}\left[\left(y^{\prime 2} + 1\right)B_x -x^{\prime}y^{\prime}B_y -x^{\prime}Bs\right]
    \end{aligned}
    \label{eq:equations_of_motion}
\end{equation}

\subsection{First-order solutions to the equations of motion}
\par Assuming the transverse velocity of the electrons is zero, the equations of motion reduce to:
\begin{equation}
    \begin{aligned}
        x^{\prime\prime} = \frac{1}{R}B_y\\
        y^{\prime\prime} = -\frac{1}{R}B_x
    \end{aligned}
    \label{eq:first_order_eqm}
\end{equation}
The integration of equation \eqref{eq:first_order_eqm} leads to:
\begin{equation}
    \begin{aligned}
        x^{\prime}(s) = \frac{1}{R}\int_{-\infty}^{s} B_y(0, 0, s^{\prime}) ds^{\prime}\\
        y^{\prime}(s) = -\frac{1}{R}\int_{-\infty}^{s}B_x(0, 0, s^{\prime}) ds^{\prime}\\
        x(s) = \frac{1}{R}\int_{-\infty}^{s}\int_{-\infty}^{s^{\prime}}B_y(0, 0, s^{\prime\prime}) ds^{\prime\prime} ds^{\prime}\\
        y(s) = -\frac{1}{R}\int_{-\infty}^{s}\int_{-\infty}^{s^{\prime}}B_x(0, 0, s^{\prime\prime}) ds^{\prime\prime} ds^{\prime}\\
    \end{aligned}
    \label{eq:field_integrals}
\end{equation} 
These are the so called field integrals. These equations are a good approximation for elements like quadrupoles, sextupoles and correctors,
 but for a well designed insertion device, the field integrals evaluated in the whole ID are close to zero, and thus the first-order solutions are not enough to describe
  the net effect of the undulator on the electron beam. In this case, it is necessary to consider the second-order solutions to the equations of motion.

\subsection{Second-order solutions to the equations of motion}

In the second-order solutions we will keep terms only to the second order of the inverse of the magnetic rigidity $1/R^2$. Looking at equations 
\eqref{eq:field_integrals} we can see that all phase space variables are proportional do $1/R$, so we will keep terms only to the first order of the
phase space variables, then the equations of motion become:

\begin{equation}
    \begin{aligned}
        x^{\prime\prime} = \frac{1}{R}\left(y^{\prime}B_s - B_y\right)\\
        y^{\prime\prime} = -\frac{1}{R}\left(B_x - x^{\prime}B_s\right)
    \end{aligned}
    \label{eq:second_order_eqm}
\end{equation}

Next, we are going to Taylor expand the magnetic field around the reference trajectory.
\begin{equation}
    \begin{aligned}
        B_y(x, y, s) = B_y(0, 0, s) + x\frac{\partial B_y}{\partial x}(0, 0, s) + y\frac{\partial B_y}{\partial y}(0, 0, s) + \mathcal{O}(2)\\
        B_x(x, y, s) = B_x(0, 0, s) + x\frac{\partial B_x}{\partial x}(0, 0, s) + y\frac{\partial B_x}{\partial y}(0, 0, s) + \mathcal{O}(2)
    \end{aligned}
    \label{eq:field_expanded}
\end{equation}
then, combining equations \eqref{eq:field_expanded} and \eqref{eq:second_order_eqm} we obtain the following form for the equations of motion:
\begin{equation}
    \begin{aligned}
        x^{\prime\prime} = -\frac{1}{R}\left[y^{\prime}B_s^{0} - B_y^{0} - x\frac{\partial B_y}{\partial x}^{0} - y\frac{\partial B_y}{\partial y}^{0}\right]\\
        y^{\prime\prime} = -\frac{1}{R}\left[B_x^{0} + x\frac{\partial B_x}{\partial x}^{0} + y\frac{\partial B_x}{\partial y}^{0} - x^{\prime}B_s^{0}\right]
    \end{aligned}
    \label{eq:second_order_eqm_simplified}
\end{equation}

where the supscript $^0$ denotes the evaluation in the reference point ($0, 0, s$). When we insert the first-order solutions (which
are valid for the electron motion inside the device, since the field integrals are close to zero only when we integrate over the whole ID), into 
equation \eqref{eq:second_order_eqm_simplified}, and integrate over the whole ID we obtain the following expressions for kick experienced by the
electron:
\begin{equation}
    \begin{aligned}
        \Delta x^{\prime} = \int_{-\infty}^{\infty}x^{\prime \prime}ds = \frac{1}{R^2}\int_{-\infty}^{\infty}\left[B_s^{0}\int_{-\infty}^{s}B_x^{0}ds^{\prime}
        +\frac{\partial B_y^0}{\partial x}\int_{-\infty}^{s}\int_{-\infty}^{s^{\prime}}B_y^{0}ds^{\prime \prime}ds^{\prime}
        -\frac{\partial B_y^0}{\partial y}\int_{-\infty}^{s}\int_{-\infty}^{s^{\prime}}B_x^{0}ds^{\prime \prime}ds^{\prime}\right]ds \\
        \Delta y^{\prime} = \int_{-\infty}^{\infty}y^{\prime \prime}ds = \frac{1}{R^2}\int_{-\infty}^{\infty}\left[B_s^{0}\int_{-\infty}^{s}B_y^{0}ds^{\prime}
        +\frac{\partial B_x^0}{\partial y}\int_{-\infty}^{s}\int_{-\infty}^{s^{\prime}}B_x^{0}ds^{\prime \prime}ds^{\prime}
        -\frac{\partial B_x^0}{\partial x}\int_{-\infty}^{s}\int_{-\infty}^{s^{\prime}}B_y^{0}ds^{\prime \prime}ds^{\prime}\right]ds
    \end{aligned}
   \label{eq:second_order_solutions_integral} 
\end{equation}

\par Equation \eqref{eq:second_order_solutions_integral} gives the kick experienced by an electron due to its wiggling motion through the ID,
which is a kickmap. However, the kickmap equation is not in its final form, it is possible to express the kickmap in terms
 of a potential called Kickmap Potential.

\subsection{Kickmaps}
\par In order to transform equation \eqref{eq:second_order_solutions_integral} in the usual form, it is necssary to make 
use of Maxwell equations and integration by parts.
First, consider the divergence of the magnetic filed in free space $\vec{\nabla} \cdot \vec{B} = 0$
\begin{equation}
    \frac{\partial B_{x}^{0}}{\partial x} + \frac{\partial B_{y}^{0}}{\partial y} + \frac{\partial B_{s}^{0}}{\partial s} = 0
    \label{eq:divergence_free}
\end{equation}

which allow us to write $B_s$ as function of the transverse components of the magnetic field:

\begin{equation}
    B_s^{0} = \int_{-\infty}^{s}\frac{\partial B_s^{0}}{\partial s^{\prime}}ds^{\prime} =
     -\int_{-\infty}^{s}\frac{\partial B_x^{0}}{\partial x}ds^{\prime} - \int_{-\infty}^{s}\frac{\partial B_y^{0}}{\partial y}ds^{\prime}
     \label{eq:Bs_integral}
\end{equation}

Now, consider the integration by parts formula:

\begin{equation}
    \int_{a}^{b}f(s)g^{\prime}(s)ds = \left[f(s)g(s)\right]_{a}^{b} - \int_{a}^{b}f^{\prime}(s)g(s)ds
    \label{eq:int_parts}
\end{equation}

then, the second term of equation \eqref{eq:second_order_solutions_integral} can be rewritten as, where it was used
also the fact that the second integral of the transverse field components are zero.

\begin{equation}
    \begin{aligned}
    \int_{-\infty}^{\infty}\left[\frac{\partial B_{y}^{0}}{\partial x}\int_{-\infty}^{s}\int_{-\infty}^{s^{\prime}}B_{y}^{0}ds^{\prime \prime}ds^
    {\prime}\right]ds = \left[\int_{-\infty}^{s}\frac{\partial B_y^{0}}{\partial x} ds^{\prime}\int_{-\infty}^{s}\int_{-\infty}^{s^\prime}
    B_{y}^{0}ds^{\prime \prime}ds^{\prime}\right]_{\infty}^{\infty} \\ 
    - \int_{-\infty}^{\infty}\left[\int_{-\infty}^{s}\frac{\partial B_y^{0}}{\partial x}ds^{\prime}\int_{-\infty}^{s}
    B_{y}^{0}ds^{\prime}\right]ds \\
    = - \int_{-\infty}^{\infty}\left[\int_{-\infty}^{s}\frac{\partial B_y^{0}}{\partial x}ds^{\prime}\int_{-\infty}^{s}
    B_{y}^{0}ds^{\prime}\right]ds
    \end{aligned}
    \label{eq:second_term}
\end{equation}

the same simplification can be done for the third term of equation \eqref{eq:second_order_solutions_integral}:
\begin{equation}
    \int_{-\infty}^{\infty}\left[\frac{\partial B_y^{0}}{\partial y}\int_{-\infty}^{s}\int_{-\infty}^{s^{\prime}}B_{x}^{0}ds^{\prime \prime}
    ds^{\prime}\right]ds = -\int_{-\infty}^{\infty}\left[\int_{-\infty}^{s}\frac{\partial B_y^{0}}{\partial y}ds^{\prime}
    \int_{-\infty}^{s}B_x^{0}ds^{\prime}\right]
    \label{eq:third_term}
\end{equation}

and considering equations \eqref{eq:Bs_integral}, \eqref{eq:second_term} and \eqref{eq:third_term}, we can rewrite the horizontal kick as:

\begin{equation}
    \begin{aligned}
        \Delta x^{\prime} = \frac{1}{R^2}\int_{-\infty}^{\infty}
        \Bigg[
            -\int_{-\infty}^{s}\frac{\partial B_x^{0}}{\partial x}ds^{\prime}
            \int_{-\infty}^{s}B_x^{0}ds^{\prime} 
            -\int_{-\infty}^{s}\frac{\partial B_y^{0}}{\partial y}ds^{\prime}
            \int_{-\infty}^{s}B_x^{0}ds^{\prime} \\
            -\int_{-\infty}^{s}\frac{\partial B_y^{0}}{\partial x}ds^{\prime}
            \int_{-\infty}^{s}B_y^{0}ds^{\prime} 
            +\int_{-\infty}^{s}\frac{\partial B_y^{0}}{\partial y}ds^{\prime}
            \int_{-\infty}^{s}B_x^{0}ds^{\prime} \Bigg] \\
        = - \frac{1}{R^2}\int_{-\infty}^{\infty}\frac{\partial}{\partial x}\Bigg[\left(\int_{-\infty}^{s}B_x^{0}ds^{\prime}\right)^2 +
        \left(\int_{-\infty}^{s}B_y^{0}ds^{\prime}\right)^2 \Bigg]ds
        \end{aligned}
        \label{eq:delta_x_prime_final}
\end{equation}

the vertical kick can also be written in the following form:
\begin{equation}
    \Delta y^{\prime} = - \frac{1}{R^2}\int_{-\infty}^{\infty}\frac{\partial}{\partial y}\Bigg[\left(\int_{-\infty}^{s}B_x^{0}ds^{\prime}\right)^2 +
        \left(\int_{-\infty}^{s}B_y^{0}ds^{\prime}\right)^2 \Bigg]ds
    \label{eq:delta_y_prime_final}
\end{equation}

Equations \eqref{eq:delta_x_prime_final} and \eqref{eq:delta_y_prime_final} are calculating the kicks experienced by an electron, with initial conditions $x, y = (0, 0)$, due
 to its wiggling motion through the ID. However, these equations are valid for any initial condition $(x, y)$,
 the set of kicks for a grid of initial conditions is a kickmap. The kickmap
can be expressed in terms of a potential $\Phi(x, y)$, called Kickmap potential, as follows:


\begin{equation}
    \Phi(x, y) = \int_{-\infty}^{\infty}\left(\int_{-\infty}^{s}B_x(x, y, s^{\prime})ds^{\prime}\right)^2 + 
    \left(\int_{-\infty}^{s}B_y(x, y, s^{\prime})ds^{\prime}\right)^2 ds
    \label{eq:kickmap_potential}
\end{equation}

\begin{equation}
    \begin{aligned}
        \Delta x^{\prime} = -\frac{1}{R^2}\frac{\partial \Phi}{\partial x}\\
        \Delta y^{\prime} = -\frac{1}{R^2}\frac{\partial \Phi}{\partial y}
    \end{aligned}
    \label{eq:kickmap_final}
\end{equation}


\subsection{Symplecticity conditions}

\par Once we obtained the kickmap in terms of the kickmap potential, we can check if the kickmap is symplectic. 

The symplecticity conditions for a 2n-D map are given by
\begin{equation}
    J^{T}SJ = S
    \label{eq:symplecticity_condition}
\end{equation}

where $S$ is the $2n \times 2n$ symplectic matrix, defined in block form as

\begin{equation}
    S
    =
    \operatorname{diag}_{k=1,\ldots,n}
    \left[
    \begin{pmatrix}
    0 & 1\\
    -1 & 0
    \end{pmatrix}
    \right].
\end{equation}

where $I_n$ denotes the $n \times n$ identity matrix. $J$ is the jacobian matrix of the map.
Considering a $4D$ map for the kickmap, the jacobian of the kickmap can be written as:


\begin{equation}
    J
    =
    \left[
    \begin{pmatrix}
    1 & 0 & 0 & 0\\
    -\frac{1}{R^2}\frac{\partial^2\phi}{\partial^2 x} & 1 & -\frac{1}{R^2}\frac{\partial^2\phi}{\partial y \partial x}  & 0\\
    0 & 0 & 1 & 0\\
    -\frac{1}{R^2}\frac{\partial^2\phi}{\partial x \partial y} & 0 & -\frac{1}{R^2}\frac{\partial^2\phi}{\partial^2 y} & 1\\
    \end{pmatrix}
    \right].
\end{equation}
using equation \eqref{eq:kickmap_final}, the symplecticity condition \eqref{eq:symplecticity_condition}
can be expressed as:

\begin{equation}
    \frac{\partial^2\phi}{\partial x \partial y} = \frac{\partial^2 \phi}{\partial y \partial x} \\
    \label{eq:symplecticity_condition_kickmap}
\end{equation}

The Schwarz theorem states that if the second partial derivatives of a function are continuous, then the mixed partial derivatives are equal.
Thus, for any kickmap potential $\Phi(x, y)$ with continuous second partial derivatives, the kickmap is symplectic, that is the case for 
sinusoidal functions and also polynoms, which are used to describe the kickmap potential.

Considering the $6D$ case, where $\Delta z = 0$ and $\Delta \delta = 0$, the symplecticity condition is also satisfied as long as
the symplecticity condition for the $4D$ case is satisfied. This can be seen by writing the jacobian matrix for the $6D$ case as:
\begin{equation}
    J = \begin{pmatrix}
        J_{4D} & 0 \\
        0 & I_2 \\
    \end{pmatrix}
\end{equation}
where $J_{4D}$ is the $4D$ jacobian. 

\subsection{Application for a sinusoidal field}

Considering now a vertical magnetic field with a sinusoidal profile
\begin{equation}
    B_y = \sum_{n} a_n \sin(n k_s s)
    \label{eq:sinusoidal_field}
\end{equation}
It is necessary to impose that the magnetic field must be zero at the entrance and exit of the device, ie, $B_y(\pm \infty) = 0$.
Then the first integral of equation \eqref{eq:kickmap_potential} is:
\begin{equation}
    I = \int_{-\infty}^{s}B_y(s^{\prime})ds^{\prime} = \sum_{n}\frac{-a_n}{nk_s}\cos(n k_s s)
    \label{eq:first_integral_sinusoidal}
\end{equation}
And the external integral of equation \eqref{eq:kickmap_potential} is:
\begin{equation}
    \Phi(x, y) = \int_{-\infty}^{\infty} I^2 ds = n_p \sum_{n}\frac{a_n^2}{2 n^2 k_s^2}\int_{\lambda/2}^{-\lambda/2}\cos(n k_s s)ds =
     \frac{L \lambda^2}{8\pi^2}\sum_{n}\frac{a_n(x, y)^2}{n^2}
    \label{eq:kickmap_potential_sinusoidal}
\end{equation}
where $L$ is the length of the device, $\lambda$ is the period of the device and $n_p$ is the number of periods.
If the horizontal component of the field is considered, the kickmap potential can be written as:
\begin{equation}
    \Phi(x, y) = \frac{L \lambda^2}{8\pi^2}\sum_{n}\frac{a_n(x, y)^2 + b_n(x, y)^2}{n^2}
    \label{eq:kickmap_potential_sinusoidal_2D}
\end{equation}
where $b_n$ are the Fourier components of the horizontal magnetic field.

This approach can be used to calculate the kickmap potential of an arbitrary undulator.
 The only information required is the set of Fourier components of the magnetic field, no prior knowledge
  of the complete field profile is necessary. Once these coefficients are obtained, the kickmap
   potential can be calculated using Eq. \eqref{eq:kickmap_final}.

To obtain the kickmap, the potential must be differentiated with respect to the transverse coordinates.
 An elegant approach is to fit the kickmap potential using a two-dimensional polynomial. The derivatives
  can then be calculated analytically by operating directly on the polynomial coefficients, from which
   the kickmap is obtained. This approach avoids numerical differentiation, which can introduce noise
    when the grid used to calculate the kickmap potential is not sufficiently fine. \cite{lnlsfac_idanalysis}

Furthermore, as discussed in the previous section, representing the kickmap potential
 as a polynomial ensures the symplecticity of the resulting kickmap, provided that the polynomial
  is sufficiently smooth. In particular, polynomials have continuous second derivatives, so Schwarz's
   theorem guarantees the equality of the mixed partial derivatives required by the symplecticity condition.

\subsection{Application for an adjustable-phase undulator}
Solving the Laplace's equation for a current free region, it is possible to find an analitical solution for the 
magnetic field of an adjustable-phase undulator, which, in the region near the central magnetic axis is given
 by the following equations:

\begin{equation}
    \begin{aligned}
        B_y = \frac{-B_0}{2}\cos{k_xx}\left[e^{-k_y y}\sin{k_s s} + e^{k_y y}\sin{(k_s s - k_s \theta)}\right]\\
        B_x = \frac{-B_0}{2}\frac{k_x}{k_y}\sin{k_xx}\left[e^{-k_y y}\sin{k_s s} - e^{k_y y}\sin{(k_s s - k_s \theta)}\right]\\
        B_s = \frac{B_0}{2}\frac{k_s}{k_y}\cos{k_xx}\left[e^{-k_y y}\cos{k_s s} - e^{k_y y}\cos{(k_s s - k_s \theta)}\right]
    \end{aligned}
    \label{eq:apu_field}
\end{equation}
where $\theta$ is the undulator phase, $k_s = 2\pi/\lambda$ is the undulator wavenumber,
$k_x = \pi/w$ is the wavenumber associated with the undulator block width $w$, and $k_y = \sqrt{k_x^2 + k_s^2}$.
The kickmap potential for an adjustable-phase undulator can be calculated by substituting equation \eqref{eq:apu_field} into equation \eqref{eq:kickmap_potential}
  and integrating over the undulator length. The final expression for the kickmap potential, considering both transverse components of the magnetic field, is given by:
  \begin{equation}
    \phi(x, y) = \frac{LB_0^2}{4}\left[\frac{1}{k_s^2}\cos{^2k_x x}\left(\cosh{2k_y y} + \cos{k_s \theta}\right) + (\frac{k_x}{k_y k_s})^2\sin^2{k_x x}\left(\cosh{2 k_y y} - \cos{k_z\theta}\right)\right]
    \label{eq:potential_apu}
\end{equation}
then the kicks can be found by differentiating the potential with respect to the transverse coordinates, as shown in equation \eqref{eq:kickmap_final}, which yields to:

\begin{equation}
    \begin{aligned}
        \Delta x^{\prime} = \frac{LB_0^2}{4R^2}\frac{k_x}{k_s^2}\sin{k_x x}\cos{k_x x}\left(\left(1-\frac{k_x^2}{k_s^2}\right)\cosh{2k_y y} + \left(1+\frac{k_x^2}{k_s^2}\right)\cos{k_s \theta}\right)\\
        \Delta y^{\prime} = -\frac{LB_0^2}{4R^2}\frac{k_y}{k_s^2}\sinh{2 k_y y}\left(\cos^2{k_x x} + \frac{k_x^2}{k_y^2}\sin{k_x x}\right)
    \end{aligned}
    \label{eq:kicks_apu}
\end{equation}
by Taylor expanding equations \ref{eq:kicks_apu} around the origin, it is possible to find the linear
 and nonlinear components of the kickmap, keeping only the linear terms, we found the tune-shift caused by the undulator.
 The expression for the kicks  in linear approximation is given by:
\begin{equation}
    \begin{aligned}
        \Delta x^{\prime} = -\frac{LB_0^2}{4R^2}\frac{k_x^2}{k_s^2}\left(1 + \cos{k_z\theta} - \frac{k_x^2}{k_y^2}+\frac{k_x^2}{k_y^2}\cos{k_s\theta}\right)x\\
        \Delta y^{\prime} = -\frac{LB_0^2}{2R^2}\frac{k_y^2}{k_s^2}y
    \end{aligned}
    \label{eq:kicks_apu_linear}
\end{equation}
here, it is worth to note that for an undulator with infinite block width, the horizontal kick is zero, since $k_x = 0$, however, even
for a finite block width, the horizontal kick is very small, since $k_x << k_y$. Even for an infinite block width, there will be a 
vertical kick, since , in this case $k_y = k_s = \frac{2 \pi}{\lambda}$. In general the ratio of the gradient of the transverse kicks 
$\frac{\Delta y^{\prime}}{\Delta y}/\frac{\Delta x^{\prime}}{\Delta x}$
is in the order of $2k_y^2/k_x^2$, which in the most cases is close to $10^2$, that will imply that the vertical focusing will be much stronger than
the horizontal.

The tune shifts introduced by a quadrupolar perturbation in the storage ring are given by: \cite{sands1970physics}
\begin{equation}
    \Delta \nu_{x, y} = -\frac{1}{4\pi} k_{x, y}(s)\beta_{x, y}(s)L
    \label{eq:tune_shift}
\end{equation}
where $k_{x, y}(s) = \frac{1}{R}\frac{dB_y}{dx}$ is the quadrupolar strength, $\beta_{x, y}(s)$ is the beta function at the location of the quadrupole and $L$ is the length of the quadrupole.
From equation \ref{eq:kicks_apu_linear}, considering only the vertical kick we can see that the quadrupolar strength introduced by the undulator is given by:
\begin{equation}
    k(s) = -\frac{B_0^2}{2R^2}\frac{k_y^2}{k_s^2}
    \label{eq:quadrupolar_strength}
\end{equation}
then, the vertical tune shift introduced by the APU is:

\begin{equation}
    \Delta \nu_y = \frac{1}{8\pi}\frac{B_0^2}{R^2}\frac{k_y^2}{k_s^2}\beta_y L
    \label{eq:vertical_tune_shift}
\end{equation}
as one can observe, the vertical tune shift, is independent of the undulator phase $\theta$.
