\section{Introduction} 
\par
This document presents a detailed overview of insertion devices tracking methods.
The first section is an introduction to the Ellaume formalism, derivations and predicted effects
on the beam. The second section presents the tracking method based on Runge-Kutta integration and compares 
results with those obtained from the Ellaume formalism. Finally, last section presents an symplectic integration method
proposed by Wu, Forest and Robin and also compares the results with the two first methods. The main goal of this document
 is to provide a complete overview of the methods used to track
  insertion devices in the SIRIUS storage ring and to compare their results.

\section{Ellaume formalism}
\subsection{Equations of motion}
Using SIRIUS coordinate system, the equations of motion for a particle in a magnetic field can be expressed as:
\begin{equation}
    \frac{d\vec{p}}{dt} = q\vec{v} \times \vec{B}
    \label{eq:lorentz_force}
\end{equation}
$\vec{p}$ is the momentum of the electron, $q= -e$ is the electron's charge, $\vec{v}$ is its velocity, and $\vec{B}$ is
the magnetic field along the electron's path.
The velocity can be written as:
\begin{equation}
    \vec{v} = \frac{dx}{dt}\hat{x} + \frac{dy}{dt}\hat{y} + \frac{ds}{dt}\hat{s} = 
    \left(\frac{ds}{dt}\right)\left(\frac{dx}{ds}\hat{x} + \frac{dy}{ds}\hat{y} + \hat{s}\right)
    \label{eq:velocity}
\end{equation}
defining
\begin{equation}
    x\prime \coloneqq \frac{dx}{ds}, \quad y\prime \coloneqq \frac{dy}{ds}
    \label{eq:prime}
\end{equation}
and assuming the electorn is ultra-relativitic, we have:
\begin{equation}
    c^2 = \vec{v}^2 = \left(\frac{ds}{dt}\right)^2(x\prime^2 + y\prime^2 + 1)
\end{equation}
then, it is found the following expression for $ds/dt$:
\begin{equation}
    \frac{ds}{dt} = \frac{c}{\sqrt{x\prime^2 + y\prime^2 + 1}}
\end{equation}
which gives the expression for the velocity:
\begin{equation}
    \vec{v} = \frac{c}{\sqrt{x\prime^2 + y\prime^2 + 1}}\left(x\prime\hat{x} + y\prime\hat{y} + \hat{s}\right)
\end{equation}
Substituting this expression into the Lorentz force equation, we have:
\begin{equation}
    \frac{d\vec{p}}{dt} = \frac{-ec}{\sqrt{x\prime^2 + y\prime^2 + 1}}\left[(y\prime B_s-B_y)\hat{x} -(x\prime B_s-B_x)\hat{y} + (x\prime By-y\prime Bx)\hat{s}\right]
    \label{eq:lorentz_force_expanded}
\end{equation}
Keeping the result of equation \eqref{eq:lorentz_force_expanded}, now let's express the momentum of a relativistic electron:
\begin{equation}
    \vec{p} = \gamma m_e \vec{v} = \frac{\gamma m_e c}{\sqrt{x\prime^2 + y\prime^2 + 1}}\left(x\prime\hat{x} + y\prime\hat{y} + \hat{s}\right)
\end{equation}
and it's time derivative:
\begin{equation}
    \frac{d\vec{p}}{dt} = \frac{ds}{dt}\frac{d\vec{p}}{ds} = \frac{\gamma m_e c^2}{\sqrt{x\prime^2 + y\prime^2 + 1}}
    \left[-(x\prime^2+y\prime^2+1)^{-3/2}(x\prime x\prime \prime + y\prime y\prime \prime)(x\prime\hat{x} + y\prime\hat{y} + \hat{s})
    + \frac{x\prime\prime\hat{x} + y\prime\prime\hat{y}}{\sqrt{x\prime^2 + y\prime^2 + 1}}\right]
\end{equation}
